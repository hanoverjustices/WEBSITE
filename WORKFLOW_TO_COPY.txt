name: Publish HJPA website
on:
  push:
    branches: [main]
  workflow_dispatch:
permissions:
  contents: read
  pages: write
  id-token: write
concurrency:
  group: hjpa-pages
  cancel-in-progress: true
jobs:
  publish:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v6
        with:
          fetch-depth: 0
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - run: python -m pip install Pillow==11.3.0
      - uses: actions/configure-pages@v5
      - name: Generate poster and slideshow indexes
        run: python scripts/build_content.py
      - uses: actions/upload-pages-artifact@v4
        with:
          path: _site
      - name: Publish
        id: deployment
        uses: actions/deploy-pages@v4
